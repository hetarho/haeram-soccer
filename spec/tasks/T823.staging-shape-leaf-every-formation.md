  ```ts
  export type StagingPhase = 'buildUp' | 'progression' | 'finalThird' | 'counter' | 'setPiece';
  export interface ShapeSide { tactic: Tactic; roles: readonly RoleId[] /* by slot, 0 = keeper */; present?: readonly boolean[] /* by slot; absent = all */ }
  export interface ShapeContext { sides: readonly [ShapeSide, ShapeSide]; ball: Vec /* pitch metres */; team: 0 | 1 /* in possession */; phase: StagingPhase }
  export const FORMATION_SHAPE: Record<Tactic, { ip: readonly (Anchor | null)[]; oop: readonly (Anchor | null)[] }>;
  export const ROLE_SHAPE: Partial<Record<RoleId, RoleShape>>;
  export function nominalSpot(ctx: ShapeContext, side: 0 | 1, slot: number): Vec;
  /** A slot's place for a ball at the centre of `cell` (the side's own attack frame), in or out of possession, phase by the cell's band. */
  export function placeFor(sides: readonly [ShapeSide, ShapeSide], side: 0 | 1, slot: number, cell: number, inPossession: boolean): Vec;
  ```
